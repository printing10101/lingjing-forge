## 二、核心转换器 `dxf/dxf_to_model.py`

### 主入口 `convert()`

```python
def convert(self, feature_result, user_id=None, source_dxf=None):
    result = ModelConversionResult()
    length = feature_result.overall_length
    width  = feature_result.overall_width
    height = feature_result.overall_height

    if length <= 0 or width <= 0:
        result.errors.append("零件外形尺寸无效，无法创建3D模型")
        return result
    if height <= 0:
        height = 10.0
        result.warnings.append("零件高度无效，使用默认值10mm")

    base = self._create_base_solid(length, width, height)   # 建外形立方体

    hole_count = 0
    for hole in feature_result.holes:
        try:
            base = self._add_hole(base, hole, height)       # 布尔减挖孔
            hole_count += 1
        except (...) as e:
            result.warnings.append(...)                     # 单孔失败不阻塞整体

    result.workplane = base
    result.hole_count = hole_count
    return result
```

### 建外形 `_create_base_solid()`

```python
def _create_base_solid(self, length, width, height):
    # XY平面建 box，底部位于 Z=0，高度沿 Z 轴正向
    base = cq.Workplane("XY").box(length, width, height, centered=(True, True, False))
    return base
```

### 挖孔 `_add_hole()`

```python
def _add_hole(self, workplane, hole, base_height):
    hole_radius = hole.diameter / 2.0
    hole_depth = hole.depth
    if hole.hole_type == "through_hole" or hole_depth <= 0:
        hole_depth = base_height * 1.2            # 通孔穿透
    hole_depth = min(hole_depth, base_height * 1.5)

    hole_solid = (cq.Workplane("XY")
        .workplane(offset=-hole_depth * 0.1)
        .center(hole.center_x, hole.center_y)
        .circle(hole_radius)
        .extrude(hole_depth * 1.2))               # 圆柱体（略穿透）

    result = workplane.cut(hole_solid)            # 布尔减
    return result
```

### 多段线轮廓建模 `convert_from_polylines()`

```python
def convert_from_polylines(self, polylines, height=10.0):
    processor = PolylineOutlineProcessor()
    outlines = processor.extract_outlines(polylines)
    outer = outlines[0]                           # 最大面积=外轮廓
    holes = [o for o in outlines[1:] if o.is_hole]

    # 1. 画外轮廓并拉伸成体
    wp = cq.Workplane("XY").moveTo(outer.vertices[0][0], outer.vertices[0][1])
    for v in outer.vertices[1:]:
        wp = wp.lineTo(v[0], v[1])
    wp = wp.close()
    base = wp.extrude(height)

    # 2. 挖掉内部孔
    for hole in holes:
        hwp = cq.Workplane("XY").moveTo(hole.vertices[0][0], hole.vertices[0][1])
        for v in hole.vertices[1:]:
            hwp = hwp.lineTo(v[0], v[1])
        hwp = hwp.close().extrude(height * 1.2)
        base = base.cut(hwp)                       # 布尔减
    return result
```
